# ── Face detection & landmarks ────────────────────────────────────────────────
# face_pipeline.py runs MediaPipe's Tasks API (FaceLandmarker, FaceDetector with
# the full-range BlazeFace model). The model files are baked into the image by the
# Dockerfile. mediapipe pulls opencv-contrib-python, which provides `cv2`: it is
# pinned below and opencv-python-headless is NOT listed, because two OpenCV
# distributions installed together overwrite each other's cv2 package.
mediapipe==1.2.3
opencv-contrib-python==4.14.0.94

# ── GLB manipulation ──────────────────────────────────────────────────────────
pygltflib==1.16.2

# ── Interpolation (TPS warp) ──────────────────────────────────────────────────
scipy>=1.16.3,<2

# ── Image processing ──────────────────────────────────────────────────────────
Pillow==13.0.0  # clears every 11.x/12.x decoder advisory; needs Python 3.11 (see Dockerfile)
numpy>=2.3,<3

# ── Background removal ────────────────────────────────────────────────────────
# [gpu] pulls in onnxruntime-gpu so rembg uses CUDA when the L4 is present.
# 2.0.86 clears every published rembg advisory (path traversal in custom model
# loading, SSRF in `rembg s`, SAM model injection; fixed in 2.0.75 and 2.0.82).
# It needs numpy 2.3 and Python 3.11, which is why mediapipe is on the Tasks API.
rembg[gpu]==2.0.86

# ── Mesh operations (UV pre-computation, runs once at build time) ─────────────
trimesh==4.12.2

# ── HTTP ──────────────────────────────────────────────────────────────────────
# requests: build-time UV-map download (precompute_uv.py) from a fixed GitHub URL.
# httpx: runtime SSRF-hardened image fetches via worker_security.fetch_remote_bytes.
requests==2.33.1
httpx==0.28.1

# ── Server ────────────────────────────────────────────────────────────────────
fastapi==0.115.0
uvicorn[standard]==0.32.0
pydantic>=2.0.0

# ── GCP ───────────────────────────────────────────────────────────────────────
google-cloud-firestore==2.19.0
google-cloud-storage==2.18.0
